cmake_minimum_required(VERSION 3.21)
project(warpSPHCore LANGUAGES NONE)
option(WARPSPHCORE_BUILD_NATIVE "Compile the native backend instead of a source/JIT wheel" OFF)
set(WARPSPHCORE_NATIVE_BACKEND "cpu" CACHE STRING "cpu, cuda or hip")
set_property(CACHE WARPSPHCORE_NATIVE_BACKEND PROPERTY STRINGS cpu cuda hip)
set(WARPSPHCORE_CPU_ARCH "portable" CACHE STRING "portable, native, or a compiler CPU architecture")
option(WARPSPHCORE_BUILD_STANDALONE "Build the framework-independent C++ library" OFF)
if(WARPSPHCORE_BUILD_STANDALONE)
  include(cmake/Standalone.cmake)
endif()
if(NOT WARPSPHCORE_BUILD_STANDALONE OR WARPSPHCORE_BUILD_NATIVE)
  install(FILES src/warpSPHCore_config.py DESTINATION .)
endif()
if(NOT WARPSPHCORE_BUILD_NATIVE)
  return()
endif()
if(NOT WARPSPHCORE_NATIVE_BACKEND MATCHES "^(cpu|cuda|hip)$")
  message(FATAL_ERROR "WARPSPHCORE_NATIVE_BACKEND must be cpu, cuda or hip")
endif()
enable_language(CXX)
find_package(Python COMPONENTS Interpreter REQUIRED)
set(native_architectures "")
if(WARPSPHCORE_NATIVE_BACKEND STREQUAL "cuda")
  if(NOT CMAKE_CUDA_ARCHITECTURES)
    message(FATAL_ERROR "Set CMAKE_CUDA_ARCHITECTURES explicitly (e.g. 90 for H100)")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(native_architectures "${CMAKE_CUDA_ARCHITECTURES}")
elseif(WARPSPHCORE_NATIVE_BACKEND STREQUAL "hip")
  if(NOT CMAKE_HIP_ARCHITECTURES)
    message(FATAL_ERROR "Set CMAKE_HIP_ARCHITECTURES explicitly")
  endif()
  enable_language(HIP)
  set(native_architectures "${CMAKE_HIP_ARCHITECTURES}")
endif()
execute_process(COMMAND "${Python_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/packaging/native_config.py"
  --backend "${WARPSPHCORE_NATIVE_BACKEND}" --cpu-arch "${WARPSPHCORE_CPU_ARCH}"
  --architectures "${native_architectures}"
  --output "${CMAKE_CURRENT_BINARY_DIR}/native-config.cmake"
  --metadata "${CMAKE_CURRENT_BINARY_DIR}/build.json"
  COMMAND_ERROR_IS_FATAL ANY)
include("${CMAKE_CURRENT_BINARY_DIR}/native-config.cmake")
add_library(warpsph_native SHARED ${NATIVE_CPU_SOURCES})
target_compile_features(warpsph_native PRIVATE cxx_std_17)
target_compile_definitions(warpsph_native PRIVATE TORCH_TARGET_VERSION=0x020a000000000000)
target_compile_options(warpsph_native PRIVATE $<$<COMPILE_LANGUAGE:CXX>:${NATIVE_CPU_FLAGS}>)
target_include_directories(warpsph_native PRIVATE ${NATIVE_INCLUDE_DIRS})
find_library(torch_cpu_library torch_cpu PATHS ${NATIVE_TORCH_LIB_DIR} NO_DEFAULT_PATH REQUIRED)
target_link_libraries(warpsph_native PRIVATE "${torch_cpu_library}")
if(NOT WARPSPHCORE_NATIVE_BACKEND STREQUAL "cpu")
  target_sources(warpsph_native PRIVATE ${NATIVE_GPU_SOURCES})
  target_compile_definitions(warpsph_native PRIVATE WITH_CUDA USE_CUDA)
  find_library(torch_gpu_library NAMES torch_cuda torch_hip PATHS ${NATIVE_TORCH_LIB_DIR} NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(warpsph_native PRIVATE "${torch_gpu_library}")
  if(WARPSPHCORE_NATIVE_BACKEND STREQUAL "cuda")
    target_compile_features(warpsph_native PRIVATE cuda_std_17)
    target_link_libraries(warpsph_native PRIVATE CUDA::cudart)
  else()
    target_compile_definitions(warpsph_native PRIVATE USE_ROCM)
    set_source_files_properties(${NATIVE_GPU_SOURCES} PROPERTIES LANGUAGE HIP)
    set_target_properties(warpsph_native PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES)
  endif()
endif()
set_target_properties(warpsph_native PROPERTIES CXX_VISIBILITY_PRESET hidden
  CUDA_VISIBILITY_PRESET hidden HIP_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES
  INSTALL_RPATH "$ORIGIN/../../../torch/lib" INSTALL_RPATH_USE_LINK_PATH FALSE)
install(TARGETS warpsph_native LIBRARY DESTINATION warpSPHCore/native/_lib)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/build.json" DESTINATION warpSPHCore/native/_lib)
